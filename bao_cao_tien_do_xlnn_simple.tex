\section{Bài toán}

\begin{frame}
\frametitle{Bài toán nghiên cứu}

\textbf{Bài toán:} hỏi đáp trên hóa đơn tiếng Việt dựa trên văn bản OCR.

\vspace{0.2cm}

\begin{block}{Dạng đầu vào và đầu ra}
\begin{itemize}
    \item Đầu vào: câu hỏi tiếng Việt và văn bản OCR của hóa đơn.
    \item Đầu ra: câu trả lời ngắn như tổng tiền, ngày bán, tên cửa hàng, địa chỉ.
    \item Mô hình: ViT5 theo dạng \texttt{question + context -> answer}.
\end{itemize}
\end{block}

\vspace{0.2cm}

\begin{alertblock}{Vấn đề}
Khi OCR bị lỗi, câu trả lời có thể sai dù câu hỏi không đổi. Vì vậy, cần đo độ suy giảm trên OCR nhiễu và kiểm tra liệu huấn luyện với dữ liệu nhiễu có giúp mô hình bền vững hơn hay không.
\end{alertblock}

\end{frame}

\section{Mục tiêu}

\begin{frame}
\frametitle{Mục tiêu và câu hỏi nghiên cứu}

\textbf{Mục tiêu chính}
\begin{enumerate}
    \item Xây dựng baseline ViT5 cho bài toán hỏi đáp hóa đơn tiếng Việt trên OCR sạch.
    \item Tạo bộ OCR nhiễu có kiểm soát (14 loại) để đo mức suy giảm hiệu năng.
    \item Huấn luyện và so sánh 2 hướng tăng độ bền: Noisy Augmentation và Consistency Regularization.
\end{enumerate}

\vspace{0.25cm}

\textbf{Câu hỏi nghiên cứu}
\begin{itemize}
    \item ViT5 suy giảm bao nhiêu khi OCR text bị nhiễu, và loại lỗi nào ảnh hưởng mạnh nhất?
    \item Noise augmentation có giúp tăng điểm trên OCR nhiễu mà vẫn giữ điểm clean không?
    \item Ràng buộc biểu diễn (consistency loss) có mang lại robustness tốt hơn augmentation đơn giản không?
\end{itemize}

\end{frame}

\begin{block}{Cách huấn luyện Noise Aug}
Với \texttt{augmentation\_ratio = 1.0}, dữ liệu train được mở rộng thành hai phần: mẫu sạch và một mẫu OCR nhiễu tương ứng. Tổng số mẫu train xấp xỉ gấp đôi baseline. Consistency dùng cùng bộ nhiễu nhưng ở dạng \textit{cặp} clean/noisy.
\end{block}

\begin{frame}
\frametitle{Kế hoạch tiếp theo}

\begin{enumerate}
    \item \textbf{Bootstrap CI trên per-sample ANLS} — hiện chỉ có điểm trung bình; cần khoảng tin cậy để biết các chênh lệch nhỏ ($<0.5$ ANLS) có ý nghĩa thống kê hay chỉ là nhiễu ngẫu nhiên.
    \item \textbf{Error-type breakdown} — tách ``sai hoàn toàn'' (ANLS=0) vs ``gần đúng'' (0$<$ANLS$<$1), làm rõ noise gây lỗi kiểu nào.
    \item Phân tích định tính: ví dụ input/output cụ thể trên các case money/mixed noise.
    \item Cân nhắc chạy thêm Adapter / RON-NACA nếu còn tài nguyên GPU.
\end{enumerate}

\vspace{0.2cm}

\footnotesize{\textit{Đã cân nhắc severity scaling (L1/L3) nhưng ưu tiên thấp hơn: cần train lại toàn bộ 3 flow (checkpoint cũ không giữ được), trong khi bootstrap CI/error breakdown mang lại độ tin cậy thống kê cao hơn với cùng 1 lần train.}}

\vspace{0.2cm}

\begin{alertblock}{Rủi ro và kiểm soát}
Synthetic noise chưa phản ánh đầy đủ OCR thật; chỉ thử nghiệm trên một seed và một model (ViT5-base). Báo cáo mô tả rõ loại nhiễu, mức nhiễu, seed và giới hạn phạm vi kết luận.
\end{alertblock}

\end{frame}

\section{Kết quả thực nghiệm}

\begin{frame}
\frametitle{Kết quả chính trên ReceiptVQA (full data, 14 loại nhiễu @ L2)}

\begin{table}
\small
\centering
\begin{tabular}{lccc}
\toprule
\textbf{Phương pháp} & \textbf{Clean} & \textbf{Avg Noisy (14)} & \textbf{Drop} \\
\midrule
ViT5 baseline & 84.11 & 82.06 & 2.05 \\
ViT5 + Noisy Aug & \textbf{85.34} & \textbf{84.23} & \textbf{1.11} \\
ViT5 + Consistency & 83.97 & 82.92 & 1.05 \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.3cm}

\textbf{Nhận xét}
\begin{itemize}
    \item Noisy Aug thắng ở cả clean (+1.23) và avg noisy (+2.17) so với baseline.
    \item Consistency cải thiện drop tương đương Noisy Aug, nhưng \textbf{clean ANLS thấp hơn cả baseline} $\Rightarrow$ dấu hiệu over-regularization.
    \item Noisy Aug là phương pháp hiệu quả nhất trong 3 phương pháp đã thử.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{money\_noise vs date\_noise: cùng tham số, khác 27$\times$}

\begin{block}{Nghịch lý}
Cả 2 loại nhiễu dùng cùng xác suất áp dụng $p=0.30$ (L2), nhưng drop chênh nhau \textbf{27$\times$} (−4.99 vs −0.19 ANLS).
\end{block}

\vspace{0.2cm}

\begin{table}
\small
\centering
\begin{tabular}{lccc}
\toprule
\textbf{Field} & \textbf{\% answer khớp} & \textbf{Drop} & \textbf{Drop / \%exposure} \\
\midrule
money/number & 48.75\% & −4.99 & 0.102 \\
date & 5.22\% & −0.19 & 0.036 \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.2cm}

\textbf{Kết luận:} exposure (tỷ lệ answer chứa field đó) — không phải xác suất nhiễu — mới là yếu tố chi phối ($9.3\times$ exposure gap). Ngay cả sau chuẩn hoá, money vẫn hại hơn $\sim2.8\times$/exposure vì số sai không có "gần đúng". Correlation prob-param vs drop trên 13 loại: $r=0.36$ (yếu).

\end{frame}

\begin{frame}
\frametitle{Vì sao Consistency không vượt được Noisy Aug?}

\begin{block}{Cả 2 đều thấy clean + noisy khi train — khác ở cách dùng}
\begin{itemize}
    \item \textbf{Noisy Aug:} clean/noisy là mẫu độc lập, chỉ tối ưu CE $\Rightarrow$ model tự do khớp dữ liệu.
    \item \textbf{Consistency:} thêm ràng buộc $h_{clean} \approx h_{noisy}$ (cosine trên vector mean-pool) $\Rightarrow$ over-regularization.
\end{itemize}
\end{block}

\vspace{0.2cm}

\textbf{Bằng chứng:} Consistency có clean ANLS (83.97) \textit{thấp hơn} baseline (84.11) — ràng buộc làm giảm cả hiệu năng trên dữ liệu sạch, không chỉ trên nhiễu.

\vspace{0.2cm}
\textbf{Kết luận:} với bài toán này, augmentation đơn giản hiệu quả hơn ràng buộc biểu diễn phức tạp.

\end{frame}

\begin{frame}
\frametitle{Hướng mở rộng (nếu còn tài nguyên GPU)}

\begin{table}
\small
\centering
\begin{tabular}{lll}
\toprule
\textbf{TN} & \textbf{Phương pháp} & \textbf{Mục tiêu kiểm tra} \\
\midrule
4 & Adapter Only & Giảm số tham số huấn luyện, giữ nền ViT5 cố định \\
5 & RON-NACA & Kết hợp adapter và consistency loss \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.2cm}

\begin{itemize}
    \item \textbf{Adapter Only:} thêm bottleneck adapter (768$\to$64$\to$768, $\sim$100K tham số) vào encoder, freeze ViT5 base.
    \item \textbf{RON-NACA:} adapter học thích nghi với noise, consistency loss giữ biểu diễn bền vững — mục tiêu là hiệu quả tài nguyên, không kỳ vọng vượt Noisy Aug.
\end{itemize}

\vspace{0.15cm}
\textbf{Trạng thái:} chưa chạy; ưu tiên thấp vì Consistency (thành phần của RON-NACA) đã cho kết quả kém hơn Noisy Aug.

\end{frame}
